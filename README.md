# Домашнее задание к работе 6

## Условие задачи
Написать программу, которая по введенной дате вычисляет, сколько дней осталось до конца месяца или года в соответствии с запросом пользователя.

## 1. Алгоритм и блок-схема

### Алгоритм
1. Начало.
2. Запросить у пользователя ввод даты (день, месяц, год).
3. Определить количество дней в введенном месяце:
   - Если месяц апрель, июнь, сентябрь или ноябрь — 30 дней.
   - Если месяц февраль — проверить год на високосность (29 или 28 дней).
   - Для остальных месяцев — 31 день.
4. Проверить корректность введенной даты (существует ли такой день в этом месяце). Если нет — вывести ошибку.
5. Запросить у пользователя выбор действия:
   - 1 — расчет до конца месяца.
   - 2 — расчет до конца года.
6. Если выбран пункт 1:
   - Вычислить разницу: (количество дней в месяце) - (введенный день).
   - Вывести результат.
7. Если выбран пункт 2:
   - Вычислить остаток дней в текущем месяце.
   - В цикле прибавить количество дней всех последующих месяцев до декабря (с учетом високосности для февраля).
   - Вывести результат.
8. Конец.

## 2. Реализация программы


int DZ() {
  
        setlocale(LC_CTYPE, "RUS");

        int d, m, y, c;//день, месяц, год
        int d_in_m = 31;//дней в месяце
        int d_l = 0;//осталось дней

        printf("Введите дату (день месяц год через пробел): ");
        scanf("%d %d %d", &d, &m, &y);

        //определяем колво дней в введенном месяце
        if (m == 4 || m == 6 || m == 9 || m == 11) {
            d_in_m = 30;
        }
        else if (m == 2) {
            //проверка на високосный год
            if ((y % 4 == 0 && y % 100 != 0) || (y % 400 == 0)) {
                d_in_m = 29;
            }
            else {
                d_in_m = 28;
            }
        }

        //существует ли такая дата
        if (m < 1 || m > 12 || d < 1 || d > d_in_m) {
            printf("Ошибка: Введена некорректная дата\n");
            return 1;
        }

        printf("\nЧто вы хотите посчитать?\n");
        printf("1 - Дней до конца месяца\n");
        printf("2 - Дней до конца года\n");
        printf("Ваш выбор: ");
        scanf("%d", &c);

        //расчеты до конца месяца
        if (c == 1) {
            d_l = d_in_m - d;
            printf("\nДо конца месяца осталось дней: %d\n", d_l);
        }
        else if (c == 2) {
            //расчеты до конца года, считаем остаток текущего месяца
            d_l = d_in_m - d;

            //потом прибавляем дни следующих месяцев
            for (int i = m + 1; i <= 12; i++) {
                if (i == 4 || i == 6 || i == 9 || i == 11) {
                    d_l += 30;
                }
                else if (i == 2) {
                    //если текущий год високосный - добавляем 29 дней за февраль
                    if ((y % 4 == 0 && y % 100 != 0) || (y % 400 == 0)) {
                        d_l += 29;
                    }
                    else {
                        d_l += 28;
                    }
                }
                else {
                    d_l += 31;
                }
            }
            printf("\nДо конца года осталось дней: %d\n", d_l);
        }
        else {
            printf("\nОшибка.\n");
        }

        return 0;
    
}

int main()

{

    DZ();
    
}
